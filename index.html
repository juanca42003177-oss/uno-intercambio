import express from "express"; import cors from "cors"; import crypto from "crypto"; import { SiweMessage } from "siwe"; const app = express(); const PORT = 3000; const FRONTEND_ORIGIN = "https://juanca42003177-oss.github.io"; const nonces = new Map(); const sessions = new Map(); app.use(cors({ origin: FRONTEND_ORIGIN, credentials: true })); app.use(express.json()); app.get("/api/nonce", (req, res) => { const nonce = crypto.randomBytes(16).toString("hex"); nonces.set(nonce, { createdAt: Date.now() }); res.json({ ok: true, nonce }); }); app.post("/api/verify-wallet", async (req, res) => { try { const { address, message, signature, nonce } = req.body; if (!address || !message || !signature || !nonce) { return res.status(400).json({ ok: false, error: "Faltan campos requeridos" }); } if (!nonces.has(nonce)) { return res.status(400).json({ ok: false, error: "Nonce inválido o ya usado" }); } const siweMessage = new SiweMessage(message); const result = await siweMessage.verify({ signature, nonce }); if (!result.success) { return res.status(401).json({ ok: false, error: "Firma inválida" }); } const verifiedAddress = siweMessage.address?.toLowerCase(); const inputAddress = address.toLowerCase(); if (verifiedAddress !== inputAddress) { return res.status(401).json({ ok: false, error: "La dirección no coincide con la firma" }); } nonces.delete(nonce); const sessionToken = crypto.randomBytes(32).toString("hex"); sessions.set(sessionToken, { address: inputAddress, createdAt: Date.now() }); res.json({ ok: true, sessionToken, address: inputAddress }); } catch (error) { res.status(500).json({ ok: false, error: error.message || "Error interno del servidor" }); } }); app.get("/api/me", (req, res) => { const auth = req.headers.authorization || ""; const token = auth.replace("Bearer ", "").trim(); if (!token || !sessions.has(token)) { return res.status(401).json({ ok: false, error: "Sesión no válida" }); } const session = sessions.get(token); res.json({ ok: true, address: session.address }); }); app.listen(PORT, () => { console.log(`Servidor corriendo en http://localhost:${PORT}`); });
